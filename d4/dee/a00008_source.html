36 unsigned long get_cost(const vector<unsigned long> &vec, unsigned long c, unsigned long f, unsigned long u, unsigned long target) {
37 unsigned long cnt = 0;
38 unsigned long cost = 0;
39 for(const auto &i: vec) {
40 if(target > i) {
41 cnt += target - i;
42 }
43 }
44 cost += cnt * u;
45 cost += (cnt / c) * f;
46 if(cnt % c != 0) {
47 cost += f;
48 }
49 return cost;
50 }
