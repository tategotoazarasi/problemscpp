40 unsigned long get_cost(const vector<unsigned long> &vec, unsigned long c, unsigned long f, unsigned long u, unsigned long target) {
41 unsigned long cnt = 0;
42 unsigned long cost = 0;
43 for(const auto &i: vec) {
44 if(target > i) {
45 cnt += target - i;
46 }
47 }
48 cost += cnt * u;
49 cost += (cnt / c) * f;
50 if(cnt % c != 0) {
51 cost += f;
52 }
53 return cost;
54 }

171 SemVer::SemVer(string str) {
172 this->str = str;
173 int l = 0;
174 int r = 0;
175 while(true) {
176 if(r >= str.length() || !isdigit(str[r])) {
177 stringstream ss;
178 for(int i = l; i < r; i++) {
179 ss << str[i];
180 }
181 if(this->ver1 == -1) {
182 ss >> this->ver1;
183 } else if(this->ver2 == -1) {
184 ss >> this->ver2;
185 } else if(this->ver3 == -1) {
186 ss >> this->ver3;
187 } else if(this->ver4 == -1) {
188 ss >> this->ver4;
189 }
190 l = r + 1;
191 }
192 if(r >= str.length() || str[r] == ' ') {
193 break;
194 }
195 r++;
196 }
197 l = r;
198 while(l < str.length() && !isdigit(str[l]))
199 l++;
200 r = l;
201 int beta = 0;

203 beta *= 10;
204 beta += str[r++] - '0';
205 this->beta = beta;

311 std::size_t PairHash::operator()(const std::pair<long long, long long> &p) const noexcept {
312 std::size_t h1 = std::hash<int>{}(p.first);
313 std::size_t h2 = std::hash<int>{}(p.second);
314 h1 ^= h2 + 0x9e3779b9u + (h1 << 6) + (h1 >> 2);

317 long long dfs(const vector<long long> &vec, long long b, long long i, unordered_map<pair<long long, long long>, long long, PairHash> &cache) {
318 auto pair = make_pair(b, i);
319 if(cache.contains(pair)) {
320 return cache[pair];
321 }
322 if(b < 0) {
323 cache[pair] = 0;
324 return 0;
325 }
326 if(i == 0) {
327 if(b % vec[i] == 0) {
328 cache[pair] = 1;
329 return 1;
330 } else {
331 cache[pair] = 0;
332 return 0;
333 }
334 }
335 long long ans = 0;
336 for(int j = 0; j <= b / vec[i]; j++) {
337 ans += dfs(vec, b - j * vec[i], i - 1, cache);
338 }
339 cache[pair] = ans;
340 return ans;
341 }
