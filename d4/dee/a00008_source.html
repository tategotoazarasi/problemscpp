37 unsigned long get_cost(const vector<unsigned long> &vec, unsigned long c, unsigned long f, unsigned long u, unsigned long target) {
38 unsigned long cnt = 0;
39 unsigned long cost = 0;
40 for(const auto &i: vec) {
41 if(target > i) {
42 cnt += target - i;
43 }
44 }
45 cost += cnt * u;
46 cost += (cnt / c) * f;
47 if(cnt % c != 0) {
48 cost += f;
49 }
50 return cost;
51 }

113 total_price += vec[cur].price;

115 new_cate.insert(vec[cur].category);

168 SemVer::SemVer(string str) {
169 this->str = str;
170 int l = 0;
171 int r = 0;
172 while(true) {
173 if(r >= str.length() || !isdigit(str[r])) {
174 stringstream ss;
175 for(int i = l; i < r; i++) {
176 ss << str[i];
177 }
178 if(this->ver1 == -1) {
179 ss >> this->ver1;
180 } else if(this->ver2 == -1) {
181 ss >> this->ver2;
182 } else if(this->ver3 == -1) {
183 ss >> this->ver3;
184 } else if(this->ver4 == -1) {
185 ss >> this->ver4;
186 }
187 l = r + 1;
188 }
189 if(r >= str.length() || str[r] == ' ') {
190 break;
191 }
192 r++;
193 }
194 l = r;
195 while(l < str.length() && !isdigit(str[l]))
196 l++;
197 r = l;
198 int beta = 0;
199 while(r < str.length() && isdigit(str[r])) {
200 beta *= 10;
201 beta += str[r++] - '0';
202 this->beta = beta;
203 }
