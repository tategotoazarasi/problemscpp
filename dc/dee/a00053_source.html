16 string Solution::maximumBinaryString(string binary) {
17 int start = 0;
18 while(binary[start] == '1') {
19 start++;
20 }
21 bool have_0 = false;
22 ;
23 int cnt_1 = 0;
24 for(int i = start; i < binary.length(); i++) {
25 if(binary[i] == '1') {
26 cnt_1++;
27 } else {
28 have_0 = true;
29 }
30 }
31 ostringstream oss;
32 for(int i = 0; i < start; i++) {
33 oss << '1';
34 }
35 for(int i = start; i < binary.length() - 1 - cnt_1; i++) {
36 oss << '1';
37 }
38 if(have_0)
39 oss << '0';
40 for(int i = 0; i < cnt_1; i++) {
41 oss << '1';
42 }
43 return oss.str();
44 }

47 namespace climbing_stairs_ii {
48 int Solution::climbStairs(int n, vector<int> &costs) {
49 vector<int> dp(n + 1, INT_MAX);
50 vector<int> new_cost(n + 1);
51 new_cost[0] = 0;
52 for(int i = 0; i < costs.size(); i++) {
53 new_cost[i + 1] = costs[i];
54 }
55 dp[0] = 0;
56 for(int i = 0; i <= n; i++) {
57 for(int j = 1; j <= 3 && i + j < n + 1; j++) {
58 dp[i + j] = min(dp[i + j], dp[i] + new_cost[i + j] + j * j);
59 }
60 }
61 return dp[n];
62 }

66 long long Solution::minimumCost(int cost1, int cost2, int costBoth, int need1, int need2) {
67 long long total_cost = 0;
68 long long min_need = 0;
69 if(cost1 + cost2 > costBoth) {
70 min_need = min(need1, need2);
71 total_cost += min_need * costBoth;
72 }
73 total_cost += (need1 - min_need) * (costBoth > cost1 ? cost1 : costBoth);
74 total_cost += (need2 - min_need) * (costBoth > cost2 ? cost2 : costBoth);
75 return total_cost;
76 }

79 namespace longest_valid_parentheses {
80 int Solution::longestValidParentheses(string s) {
81 int ans = 0;
82 vector<Node *> vec(s.length());
83 for(int i = 0; i < s.length(); i++) {
84 vec[i] = new Node();
85 vec[i]->c = s[i];
86 vec[i]->pos = i;
87 vec[i]->lend = nullptr;
88 }
89 for(int i = 1; i < vec.size(); i++) {
90 if(vec[i]->c == ')') {
91 if(vec[i - 1]->lend == nullptr && vec[i - 1]->c == '(') {
92 if(i - 2 > 0 && vec[i - 2]->lend != nullptr) {
93 vec[i]->lend = vec[i - 2]->lend;
94 vec[i - 1]->lend = vec[i - 2]->lend;
95 } else {
96 vec[i]->lend = vec[i - 1];
97 vec[i - 1]->lend = vec[i - 1];
98 }
99 } else if(vec[i - 1]->lend != nullptr && vec[i - 1]->lend->pos - 1 >= 0 && vec[vec[i - 1]->lend->pos - 1]->c == '(') {
100 vec[i]->lend = vec[vec[i - 1]->lend->pos - 1];
101 vec[vec[i - 1]->lend->pos - 1]->lend = vec[vec[i - 1]->lend->pos - 1];
102 if(vec[i - 1]->lend->pos - 2 >= 0 && vec[vec[i - 1]->lend->pos - 2]->lend != nullptr) {
103 vec[i]->lend = vec[vec[i - 1]->lend->pos - 2]->lend;
104 vec[vec[i - 1]->lend->pos - 1]->lend = vec[vec[i - 1]->lend->pos - 2]->lend;
105 }
106 }
107 if(vec[i]->lend != nullptr) {
108 ans = max(ans, vec[i]->pos - vec[i]->lend->pos + 1);
109 }
110 }
111 }
112 for(int i = 0; i < vec.size(); i++) {
113 delete vec[i];
114 }
115 return ans;
116 }

120 int Solution::longestMonotonicSubarray(vector<int> &nums) {
121 int inc = 1;
122 int dec = 1;

124 for(int i = 1; i < nums.size(); i++) {
125 if(nums[i] > nums[i - 1]) {
126 current++;
127 inc = max(inc, current);

129 current = 1;

134 if(nums[i] < nums[i - 1]) {
135 current++;
136 dec = max(dec, current);
137 } else {
138 current = 1;
139 }
140 }
141

147 bool Solution::valid(const string &s, int a, int b) {
148 a++;
149 if(b > a && s[a] == '0') {
150 return false;
151 }
152 int num = 0;
153 while(a <= b) {
154 num *= 10;
155 num += s[a] - '0';
156 a++;
157 }
158 return num <= 255;
159 }
160 void Solution::dfs(const string &s, int cur, int remain, vector<string> &res, array<int, 3> his) {
161 if(remain == 0 && valid(s, cur, s.length() - 1)) {
162 ostringstream oss;
163 int his_cur = 0;
164 for(int i = 0; i < s.size(); i++) {
165 oss << s[i];
166 if(his_cur < 3 && his[his_cur] == i) {
167 his_cur++;
168 oss << '.';
169 }
170 }
171 res.push_back(oss.str());
172 } else {
173 for(int i = cur + 1; i <= cur + 3 && i < s.length(); i++) {
174 int rest = s.length() - i - 1;
175 if((3 * remain < rest) || remain > rest) {
176 continue;
177 }
178 if(valid(s, cur, i)) {
179 array<int, 3> new_his = his;
180 new_his[3 - remain] = i;
181 dfs(s, i, remain - 1, res, new_his);
182 }
183 }
184 }

186 vector<string> Solution::restoreIpAddresses(string s) {
187 if(s.length() > 12) {
188 return {};
189 }
190 vector<string> ans = {};
191 array<int, 3> his = {-1, -1, -1};
192 dfs(s, -1, 3, ans, his);
193 return ans;
