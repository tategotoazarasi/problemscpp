106 class Frame {
107 public:
108 int i;
109 int j;
110 int steps;
111 int dist;
112 Frame(int i, int j, int steps, int dist): i(i), j(j), steps(steps), dist(dist) {};
113 bool operator==(const Frame &f) const {
114 return this->i == f.i && this->j == f.j;
115 }
116 bool operator<(const Frame &f) const {
117 if(this->steps != f.steps) {
118 return this->steps > f.steps;
119 } else {
120 return this->dist > f.dist;
121 }
122 }
